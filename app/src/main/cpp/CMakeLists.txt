cmake_minimum_required(VERSION 3.22)
option(LIP_USE_BARK "Use the local Bark Pro engine" ON)
if(LIP_USE_BARK)
    project(lip_bark LANGUAGES C CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
    # Production target: Android16+ ARM64, automatic eligible Vulkan GPU placement
    # and checked same-model CPU fallback. No UI backend selector or NPU claim.
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Android" OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
        message(FATAL_ERROR "LIP_BARK_REQUIRES_ANDROID_ARM64")
    endif()
    if(DEFINED ANDROID_PLATFORM_LEVEL)
        set(_lip_api "${ANDROID_PLATFORM_LEVEL}")
    else()
        set(_lip_api "${CMAKE_SYSTEM_VERSION}")
    endif()
    if(NOT _lip_api MATCHES "^[0-9]+$" OR _lip_api LESS 36)
        message(FATAL_ERROR "LIP_BARK_REQUIRES_ANDROID_16_API36")
    endif()
    # Missing build inputs fail, never silently fetch or produce CPU-only success.
    set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
    set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(STARLING_GGML_SHARED OFF CACHE BOOL "" FORCE)
    set(STARLING_GGML_VULKAN ON CACHE BOOL "" FORCE)
    set(STARLING_GGML_CUDA OFF CACHE BOOL "" FORCE)
    set(STARLING_GGML_METAL OFF CACHE BOOL "" FORCE)
    set(STARLING_GGML_HIP OFF CACHE BOOL "" FORCE)
    set(GGML_CPU ON CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    set(GGML_OPENMP_FETCH OFF CACHE BOOL "" FORCE)
    set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
    set(STARLING_BENCH OFF CACHE BOOL "" FORCE)
    set(LIP_VULKAN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/vulkan")
    include("${LIP_VULKAN_ROOT}/verify_headers.cmake")
    lip_verify_vulkan_headers("${LIP_VULKAN_ROOT}")
    set(LIP_VULKAN_HEADERS_INCLUDE_DIR "${LIP_VULKAN_ROOT}/include")
    include(cmake/LipVulkanToolchain.cmake)
    lip_prepare_android_vulkan()
    find_package(Vulkan COMPONENTS glslc REQUIRED)
    find_package(SPIRV-Headers CONFIG REQUIRED)
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
    set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
    set(GGML_BLAS OFF CACHE BOOL "" FORCE)
    set(STARLING_GGML_TESTS OFF CACHE BOOL "" FORCE)
    set(STARLING_QUANTIZE OFF CACHE BOOL "" FORCE)
    set(STARLING_SERVE OFF CACHE BOOL "" FORCE)
    set(STARLING_FAST OFF CACHE BOOL "" FORCE)
    set(STARLING_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/starling")
    add_subdirectory("${STARLING_ROOT}/backends/native" starling_native)
    if(NOT TARGET ggml-vulkan)
        message(FATAL_ERROR "LIP_BARK_COMPILED_VULKAN_BACKEND_REQUIRED")
    endif()
    # Upstream finds this package without propagating its header include target.
    target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
    add_library(lip_bark SHARED bark_jni.cpp)
    target_include_directories(lip_bark PRIVATE "${STARLING_ROOT}/cpp")
    target_link_libraries(lip_bark PRIVATE starling_ggml)
else()
