option(LIP_USE_BARK "Use the local Bark Pro engine" ON)
if(LIP_USE_BARK)
    project(lip_bark LANGUAGES C CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
    set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
    set(GGML_BLAS OFF CACHE BOOL "" FORCE)
    set(STARLING_GGML_TESTS OFF CACHE BOOL "" FORCE)
    set(STARLING_QUANTIZE OFF CACHE BOOL "" FORCE)
    set(STARLING_SERVE OFF CACHE BOOL "" FORCE)
    set(STARLING_FAST OFF CACHE BOOL "" FORCE)
    set(STARLING_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/starling")
    add_subdirectory("${STARLING_ROOT}/backends/native" starling_native)
    add_library(lip_bark SHARED bark_jni.cpp)
    target_include_directories(lip_bark PRIVATE "${STARLING_ROOT}/cpp")
    target_link_libraries(lip_bark PRIVATE starling_ggml)
